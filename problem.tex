\noindent\textbf{18.（15分）} 已知函数 $f(x)=\ln \dfrac{x}{2-x}+ax+b(x-1)^3$.
\begin{enumerate}[label=(\arabic*)]
    \item 若 $b=0$，且 $f'(x)\ge 0$，求 $a$ 的最小值；
    \item 证明：曲线 $y=f(x)$ 是中心对称图形；
    \item 若 $f(x)>-2$ 当且仅当 $1<x<2$，求 $b$ 的取值范围。
\end{enumerate}

\vspace{0.5cm}
\noindent\textbf{解：}
函数定义域：$x\in(0,2)$。

\begin{enumerate}[label=(\arabic*)]
\item 当 $b=0$，$f(x)=\ln\frac{x}{2-x}+ax=\ln x-\ln(2-x)+ax$。
求导：
\[
f'(x)=\frac1x+\frac{1}{2-x}+a
=\frac{2}{x(2-x)}+a.
\]
分母 $x(2-x)=-x^2+2x$，在 $(0,2)$ 上最大值为 $1$（$x=1$ 处），故
\[
0<x(2-x)\le 1 \implies \frac{2}{x(2-x)}\ge 2.
\]
$f'(x)\ge 0$ 对所有 $x\in(0,2)$ 恒成立，即
\[
a+\frac{2}{x(2-x)}\ge 0.
\]
只需最小值满足：$a+2\ge 0 \implies a\ge -2$。
所以 $a$ 的最小值为 $\boldsymbol{-2}$。

\item 中心对称判定：验证 $f(1+u)+f(1-u)$ 为常数。
\[
\begin{aligned}
f(1+u)&=\ln\frac{1+u}{1-u}+a(1+u)+b u^3,\\
f(1-u)&=\ln\frac{1-u}{1+u}+a(1-u)+b (-u)^3.
\end{aligned}
\]
两式相加：
\[
\begin{aligned}
f(1+u)+f(1-u)
&=\ln\left(\frac{1+u}{1-u}\cdot\frac{1-u}{1+u}\right)+a(1+u+1-u)+b\left(u^3-u^3\right)\\
&=\ln 1 + 2a \\
&=2a.
\end{aligned}
\]
即 $f(1+u)+f(1-u)=2a$，曲线 $y=f(x)$ 关于点 $\boldsymbol{(1,a)}$ 中心对称。
因此曲线是中心对称图形。

\item 条件：$f(x)>-2$ 当且仅当 $1<x<2$。
由对称性：$f(1+u)+f(1-u)=2a$。
$x=1$ 时，$f(1)=a$。边界条件推出 $f(1)=-2$，即 $\boldsymbol{a=-2}$。

令 $t=x-1$，$x=1+t,\ t\in(-1,1)$，代入：
\[
f(1+t)=\ln\frac{1+t}{1-t}-2(1+t)+bt^3.
\]
条件转化：当 $t>0$ 时，$f(1+t)>-2$。
\[
\ln\frac{1+t}{1-t}-2(1+t)+bt^3 > -2,
\]
化简，记
\[
g(t)=\ln\frac{1+t}{1-t}-2t+bt^3>0,\quad t\in(0,1).
\]
求导：
\[
\begin{aligned}
g'(t)&=\frac{2}{1-t^2}-2+3bt^2
=\frac{2t^2}{1-t^2}+3bt^2 \\
&=t^2\left( \frac{2}{1-t^2}+3b \right).
\end{aligned}
\]
$t\in(0,1)$ 时 $t^2>0$，$g'(t)$ 的符号由 $h(t)=\dfrac{2}{1-t^2}+3b$ 决定。
$h(t)$ 在 $(0,1)$ 单调递增，$h(0)=2+3b$。

\begin{itemize}
    \item 若 $2+3b\ge 0$，即 $b\ge -\dfrac23$：
    $h(t)\ge h(0)\ge0$，$g'(t)\ge0$，$g(t)$ 在 $(0,1)$ 单调递增。
    $g(0)=0$，所以 $t>0$ 时 $g(t)>g(0)=0$，满足条件。
    \item 若 $b<-\dfrac23$：
    $h(0)=2+3b<0$，$t\to1^-$ 时 $h(t)\to+\infty$，存在 $t_0\in(0,1)$，$h(t_0)=0$。
    当 $0<t<t_0$，$h(t)<0$，$g'(t)<0$，$g(t)$ 在 $(0,t_0)$ 递减，$g(t)<g(0)=0$，矛盾。
\end{itemize}

综上，$b$ 的取值范围是 $\boldsymbol{\left[-\dfrac{2}{3},+\infty\right)}$。
\end{enumerate}
